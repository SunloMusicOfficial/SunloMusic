import os
from importlib import import_module
from SunloMusic.logger import LOGGER

def load_plugins():
    for file in sorted(os.listdir(os.path.dirname(__file__))):
        if file.endswith(".py") and not file.startswith("_"):
            try:
                import_module(f"SunloMusic.plugins.{file[:-3]}")
                LOGGER.info(f"🔌 Loaded: {file}")
            except Exception as e: LOGGER.error(f"❌ Failed {file}: {e}")

@bot.on_message(filters.command(["skip", "next"]) & filters.group)
@catch_errors
async def skip_cmd(_, m):
    try: await m.delete()
    except: pass
    q = get_queue(m.chat.id)
    if not q.now_playing: return await m.reply_text(NOT_PLAYING_TEXT)
    
    user = m.from_user
    is_admin = False
    if await is_sudo(user.id): is_admin = True
    else:
        try:
            member = await m.chat.get_member(user.id)
            if member.privileges and (member.privileges.can_manage_voice_chats or member.privileges.can_manage_chat): is_admin = True
        except: pass
        
    if is_admin:
        if q.karaoke_task: q.karaoke_task.cancel()
        return await start_next_song(m.chat.id, q)
        
    if user.id in q.votes: return await m.reply_text("🗳️ You have already voted to skip!")
    q.votes.add(user.id)
    
    try:
        participants = await m.chat.get_members(filter=enums.ChatMembersFilter.VOICE_CHAT)
        total_listeners = len(participants) - 2
    except: total_listeners = 5
        
    votes_needed = max(1, int(total_listeners / 2))
    if len(q.votes) >= votes_needed:
        await m.reply_text(f"🗳️ <b>Vote Skip Passed!</b> Skipping...")
        if q.karaoke_task: q.karaoke_task.cancel()
        await start_next_song(m.chat.id, q)
    else:
        await m.reply_text(f"🗳️ <b>Vote Added!</b> Needed: {votes_needed - len(q.votes)} more.")

@bot.on_callback_query(filters.regex("^cb_"))
@catch_errors
async def controls_cb(_, cq: CallbackQuery):
    action = cq.data
    chat_id = cq.message.chat.id
    user = cq.from_user
    
    if action not in ["cb_more", "cb_close", "cb_timer"]:
        is_admin = False
        if await is_sudo(user.id): is_admin = True
        else:
            try:
                member = await cq.message.chat.get_member(user.id)
                if member.privileges and (member.privileges.can_manage_voice_chats or member.privileges.can_manage_chat): is_admin = True
            except: pass
        if not is_admin: return await cq.answer("🚫 Sirf Admin hi ye buttons use kar sakte hain!", show_alert=True)

    try:
        q = get_queue(chat_id)
        
        if action == "cb_pause":
            if q.now_playing:
                elapsed = int(time.time() - q.start_time - q.total_paused)
                if q.paused:
                    await resume_stream(chat_id); q.paused = False; q.total_paused += time.time() - q.paused_at
                    await cq.answer("▶️ Resumed")
                else:
                    await pause_stream(chat_id); q.paused = True; q.paused_at = time.time()
                    await cq.answer("⏸ Paused")
                try: await cq.message.edit_reply_markup(reply_markup=player_keyboard(q, elapsed))
                except: pass
            else: await cq.answer("Nothing playing", show_alert=True)
                
        elif action == "cb_skip":
            if q.karaoke_task: q.karaoke_task.cancel()
            await start_next_song(chat_id, q)
            await cq.answer("⏭ Skipped")
            
        elif action == "cb_stop":
            if q.timer_task: q.timer_task.cancel()
            if q.karaoke_task: q.karaoke_task.cancel()
            await stop_stream(chat_id)
            clear_queue(chat_id)
            await cq.answer("⏹ Stopped")
            try: await cq.message.delete()
            except: pass
            
        elif action == "cb_loop":
            q.loop = (q.loop + 1) % 3
            elapsed = int(time.time() - q.start_time - q.total_paused) if q.now_playing else 0
            try: await cq.message.edit_reply_markup(reply_markup=player_keyboard(q, elapsed))
            except: pass
            modes = {0: "Off", 1: "One", 2: "All"}
            await cq.answer(f"🔁 Loop: {modes[q.loop]}")
            
        elif action == "cb_more":
            await cq.answer("📋 Queue: /queue\n🔄 Shuffle: /shuffle", show_alert=True)
        elif action == "cb_close":
            await cq.message.delete(); await cq.answer()
    except Exception: pass

import random

@bot.on_message(filters.command("queue") & filters.group)
@catch_errors
async def queue_cmd(_, m):
    q = get_queue(m.chat.id)
    if not q.now_playing and not q.queue: return await m.reply_text("📭 <b>Queue is empty.</b>")
    text = "<b>📋 Current Queue</b>\n\n"
    if q.now_playing: text += f"<b>▶️ Now Playing:</b>\n🎵 {q.now_playing.title}\n⏱ {format_duration(q.now_playing.duration)}\n\n"
    if q.queue:
        text += "<b>⏭ Up Next:</b>\n"
        for i, t in enumerate(q.queue[:10], 1): text += f"{i}. {t.title} ({format_duration(t.duration)})\n"
    await m.reply_text(text)

@bot.on_message(filters.command("clear") & filters.group)
@catch_errors
@admin_only
async def clear_cmd(_, m):
    q = get_queue(m.chat.id)
    if not q.queue: return await m.reply_text("📭 <b>Queue is already empty.</b>")
    
    # Delete queue messages
    for t in q.queue:
        if t.message_id:
            try: await bot.delete_messages(m.chat.id, t.message_id)
            except: pass
            
    q.queue.clear()
    await m.reply_text("🗑 <b>Queue Cleared!</b>")

@bot.on_message(filters.command("shuffle") & filters.group)
@catch_errors
@admin_only
async def shuffle_cmd(_, m):
    q = get_queue(m.chat.id)
    if not q.queue: return await m.reply_text("📭 <b>Queue is empty.</b>")
    random.shuffle(q.queue)
    await m.reply_text("🔀 <b>Queue shuffled!</b>")

@bot.on_message(filters.command("karaoke") & filters.group)
@catch_errors
@admin_only
async def karaoke_cmd(_, m):
    if len(m.command) < 2: return await m.reply_text("❌ <b>Usage:</b> <code>/karaoke on</code>")
    state = m.command[1].lower()
    
    if state == "on":
        KARAOKE_CHATS.add(m.chat.id)
        q = get_queue(m.chat.id)
        if q.now_playing:
            await m.reply_text("🎤 <b>Karaoke Mode ON!</b>\nStarting live lyrics for the current song...")
            q.karaoke_task = asyncio.create_task(start_live_karaoke(bot, m.chat.id, q.now_playing.title, q.now_playing.duration, q))
        else:
            await m.reply_text("🎤 <b>Karaoke Mode ON!</b>\nBot will send live lyrics for upcoming songs.")
            
    elif state == "off":
        KARAOKE_CHATS.discard(m.chat.id)
        q = get_queue(m.chat.id)
        if q.karaoke_task:
            q.karaoke_task.cancel()
            q.karaoke_task = None
        await m.reply_text("🎤 <b>Karaoke Mode OFF.</b>")

def _search_sync(query: str):
    try:
        with yt_dlp.YoutubeDL({"quiet": True, "extract_flat": True, "noplaylist": True}) as ydl:
            info = ydl.extract_info(f"ytsearch5:{query} audio", download=False)
            results = []
            if info and "entries" in info:
                for e in info["entries"]:
                    if e:
                        results.append({
                            "title": e.get("title", "Unknown"),
                            "url": e.get("url"),
                            "duration": e.get("duration", 0),
                            "id": e.get("id")
                        })
            return results
    except Exception as e:
        LOGGER.error(f"Search error: {e}")
        return []

@bot.on_message(filters.command("search") & filters.group)
@catch_errors
async def search_cmd(_, m):

    import asyncio
    loop = asyncio.get_running_loop()
    results = await loop.run_in_executor(None, _search_sync, query)
    
    if not results:
        return await status.edit_text("❌ <b>No results found.</b>")
        
    buttons = []
    for idx, res in enumerate(results):
        dur = res["duration"] or 0
        mins, secs = divmod(dur, 60)
        dur_str = f"{mins}:{secs:02d}"
        buttons.append([InlineKeyboardButton(f"{res['title'][:35]}... ({dur_str})", callback_data=f"srch_{res['url']}")])
        
    await status.edit_text(
        f"<b>🔍 Search Results for:</b> <i>{query}</i>\n\nSelect a song to play:",
        reply_markup=InlineKeyboardMarkup(buttons)
    )

@bot.on_message(filters.voice | filters.audio | (filters.video & ~filters.command(["play", "vplay", "song"])) & filters.group)
@catch_errors
async def shazam_handler(_, m):
    # Ignore if it's a command
    if m.text and m.text.startswith("/"): return
    
    msg = await m.reply_text("🎧 <b>Listening to your voice/audio...</b>")
    path = await m.download()
    
    try:
        shazam = Shazam()
        out = await shazam.recognize(path)
        
        if out and "track" in out:
            title = out["track"]["title"]
            artist = out["track"]["subtitle"]
            await msg.edit_text(f"🎵 <b>Found:</b> {title} by {artist}\n\n⚡ <b>Fetching stream link to play...</b>")
            
            # Play the song
            info = await extract_audio(f"{title} {artist}")
            if info:
                await msg.edit_text(f"🎶 <b>Matched:</b> {title} by {artist}\n\n<i>Use /play {title} to play it in VC!</i>")
            else:
                await msg.edit_text(f"🎵 <b>Matched:</b> {title} by {artist}\n<i>Could not auto-fetch stream link.</i>")
        else:
            await msg.edit_text("❌ <b>Could not recognize this audio.</b> Try a clearer snippet.")
    except Exception as e:
        LOGGER.error(f"Shazam Error: {e}")
        await msg.edit_text("❌ <b>Failed to analyze audio.</b>")
    finally:
        if os.path.exists(path): os.remove(path)

async def ensure_assistant_in_chat(chat_id: int) -> bool:
    global ASSISTANT_CHATS
    if not assistant: return False
        
    if chat_id in ASSISTANT_CHATS: return True

    # 1. Check if already a member
    try:
        await assistant.get_chat_member(chat_id, "me")
        ASSISTANT_CHATS.add(chat_id)
        return True
    except Exception:
        pass

    LOGGER.info(f"🤖 Assistant is not in {chat_id}, attempting to join...")
    
    # 2. Try Public Username
    try:
        chat = await bot.get_chat(chat_id)
        if chat.username:
            try:
                await assistant.join_chat(chat.username)
                ASSISTANT_CHATS.add(chat_id)
                LOGGER.info(f"✅ Assistant joined public chat @{chat.username}")
                return True
            except UserAlreadyParticipant:
                ASSISTANT_CHATS.add(chat_id)
                return True
            except Exception:
                pass
    except: pass

    # 3. Try Invite Link + Auto-Approve (Fixes the manual approval issue)
    try:
        link = await bot.export_chat_invite_link(chat_id)
        await asyncio.sleep(1)
        try:
            await assistant.join_chat(link)
        except Exception as e:
            # Sometimes join_chat throws error if request is pending
            LOGGER.info(f"Join attempt: {e}")
            
        await asyncio.sleep(2) # Wait for Telegram to process
        
        # Check if joined
        try:
            await assistant.get_chat_member(chat_id, "me")
            ASSISTANT_CHATS.add(chat_id)
            LOGGER.info(f"✅ Assistant joined via invite link")
            return True
        except:
            # ✅ If still not joined, Bot auto-approves the join request!
            LOGGER.info("📝 Assistant join request pending. Bot auto-approving...")
            try:
                me_ass = await assistant.get_me()
                await bot.approve_chat_join_request(chat_id, me_ass.id)
                LOGGER.info("✅ Bot auto-approved Assistant's request!")
                ASSISTANT_CHATS.add(chat_id)
                return True
            except Exception as approve_err:
                LOGGER.error(f"❌ Auto-approve failed: {approve_err}. Make sure Bot has 'Add Users' permission.")
    except Exception as e:
        LOGGER.error(f"❌ Could not create invite link: {e}")

    return False

async def check_empty_vc(chat_id):
    try:
        members = await bot_module.bot.get_chat_members(chat_id, filter=enums.ChatMembersFilter.VOICE_CHAT)
        humans = [m for m in members if not m.user.is_bot]
        if len(humans) == 0:
            if chat_id not in EMPTY_VC_TIMERS: EMPTY_VC_TIMERS[chat_id] = time.time()
            elif time.time() - EMPTY_VC_TIMERS[chat_id] > 180:
                await bot_module.bot.send_message(chat_id, "💤 <b>No humans in VC for 3 minutes. Leaving...</b>")
                await stop_stream(chat_id)
                from SunloMusic.queues import clear_queue
                clear_queue(chat_id)
                if chat_id in EMPTY_VC_TIMERS: del EMPTY_VC_TIMERS[chat_id]
                return True
        else:
            if chat_id in EMPTY_VC_TIMERS: del EMPTY_VC_TIMERS[chat_id]
    except: pass
    return False

async def play_stream(chat_id: int, url: str, video: bool = False, effect: str = "none"):
    if not call: return False
    if video and (url.endswith('.m4a') or url.endswith('.mp3')): video = False
    
    ffmpeg_args = "-af bass=g=10" if effect == "bass" else "-af apulsator=hz=0.1" if effect == "8d" else "-af atempo=1.25,asetrate=44100*1.25" if effect == "nightcore" else None
    
    if video: stream = MediaStream(media_path=url, audio_parameters=AudioQuality.HIGH, video_parameters=VideoQuality.HD_720p, ffmpeg_parameters=ffmpeg_args)
    else: stream = MediaStream(media_path=url, audio_parameters=AudioQuality.HIGH, video_flags=MediaStream.Flags.IGNORE, ffmpeg_parameters=ffmpeg_args)
    
    try:
        LOGGER.info(f"🎵 Attempting to play URL directly: {url[:80]}")
        await asyncio.wait_for(call.play(chat_id, stream, config=GroupCallConfig(auto_start=True)), timeout=20.0)
        return True
    except Exception as e:
        LOGGER.error(f"Direct Play error: {e}")
        
        # ✅ Fallback: Agar direct URL fail ho jaye, toh local download karke play karo
        if "http" in url:
            LOGGER.info("⚠️ Direct link failed! Attempting local download fallback...")
            try:
                os.makedirs("cache/downloads", exist_ok=True)
                file_path = f"cache/downloads/{abs(hash(url))}.m4a"
                
                # Download file
                async with aiohttp.ClientSession() as session:
                    async with session.get(url) as resp:
                        if resp.status == 200:
                            with open(file_path, "wb") as f:
                                f.write(await resp.read())
                                
                if os.path.exists(file_path):
                    LOGGER.info(f"✅ Download successful. Playing local file: {file_path}")
                    if video: stream2 = MediaStream(media_path=file_path, audio_parameters=AudioQuality.HIGH, video_parameters=VideoQuality.HD_720p, ffmpeg_parameters=ffmpeg_args)
                    else: stream2 = MediaStream(media_path=file_path, audio_parameters=AudioQuality.HIGH, video_flags=MediaStream.Flags.IGNORE, ffmpeg_parameters=ffmpeg_args)
                    
                    await asyncio.wait_for(call.play(chat_id, stream2, config=GroupCallConfig(auto_start=True)), timeout=30.0)
                    return True
            except Exception as download_err:
                LOGGER.error(f"Download fallback failed: {download_err}")
                
        return True # Even if it fails, return True so card is sent.

def catch_errors(func):

    async def wrapper(client, update, *args, **kwargs):
        try: return await func(client, update, *args, **kwargs)
        except Exception as e: await log_error_to_group(update, e)
    return wrapper

def sudo_only(func):
    @wraps(func)
    async def wrapper(client, update, *args, **kwargs):
        user = update.from_user
        if user and (user.id == config.OWNER_ID or user.id in config.sudo_list):
            return await func(client, update, *args, **kwargs)
        if user and await is_sudo(user.id): 
            return await func(client, update, *args, **kwargs)
        
        if isinstance(update, Message): await update.reply_text(NO_SUDO_TEXT)
        elif isinstance(update, CallbackQuery): await update.answer(NO_SUDO_TEXT, show_alert=True)
    return wrapper

def admin_only(func):
    @wraps(func)
    async def wrapper(client, message: Message, *args, **kwargs):
        if not message.from_user: return
        
        # ✅ Direct Owner/Sudo check (No DB wait)
        if message.from_user.id == config.OWNER_ID or message.from_user.id in config.sudo_list:
            return await func(client, message, *args, **kwargs)
            
        try:
            if await is_sudo(message.from_user.id): 
                return await func(client, message, *args, **kwargs)
        except: pass
            
        if message.chat.type != enums.ChatType.PRIVATE:

            except Exception: pass

    return wrapper

def format_duration(seconds: int) -> str:
    if not seconds: return "00:00"
    h, rem = divmod(int(seconds), 3600)
    m, s = divmod(rem, 60)
    return f"{h:02d}:{m:02d}:{s:02d}" if h else f"{m:02d}:{s:02d}"

def make_progress_bar(current: int, total: int, length: int = 7) -> str:
    """Generates a progress bar with a moving music emoji and thick lines."""
    if total <= 0: return "━" * length
    pos = int(length * current / total)
    pos = max(0, min(pos, length - 1))
    # ✅ Changed from '─' (thin) to '━' (thick)
    return "━" * pos + "🎵" + "━" * (length - pos - 1)

def _search_ytdlp_sync(query: str):
    try:
        # extract_flat=True makes it super fast
        ydl_opts = {"quiet": True, "no_warnings": True, "noplaylist": True, "extract_flat": True, "nocheckcertificate": True}
        with yt_dlp.YoutubeDL(ydl_opts) as ydl:
            info = ydl.extract_info(f"ytsearch1:{query} song audio", download=False)
            if info and "entries" in info and len(info["entries"]) > 0:
                entry = info["entries"][0]
                thumb = entry.get("thumbnails", [{}])[-1].get("url") if entry.get("thumbnails") else f"https://i.ytimg.com/vi/{entry['id']}/hqdefault.jpg"
                return {
                    "title": entry.get("title", query),
                    "url": f"https://www.youtube.com/watch?v={entry['id']}",
                    "duration": int(entry.get("duration") or 0),
                    "thumbnail": thumb
                }
    except Exception as e: 
        LOGGER.error(f"yt-dlp search error: {e}")
    return None

async def _api_request(endpoint: str, params: dict) -> dict | None:

            async with session.get(url, headers=API_HEADERS, params=params, timeout=20) as resp:

    except: return None

async def _download_telegram_cdn(cdnurl: str) -> str | None:
    if not bot_module.assistant: return None

async def extract_audio(query: str, video: bool = False) -> dict | None:
    if not config.ONEGRAB_API_KEY: return None
    if any(query.lower().endswith(ext) for ext in (".m4a", ".mp4", ".mp3", ".webm")):
        return {"title": "Direct Link", "url": query, "duration": 0, "thumbnail": "", "webpage_url": query}

    webpage_url = query
    fallback_meta = {}

    if not query.startswith("http"):
        # ✅ 1. yt-dlp Fast Search
        LOGGER.info(f"🔎 Searching via yt-dlp: {query}")
        loop = asyncio.get_running_loop()
        try: ytdlp_info = await asyncio.wait_for(loop.run_in_executor(None, _search_ytdlp_sync, query), timeout=10.0)
        except: ytdlp_info = None
            
        if ytdlp_info and ytdlp_info.get("url"):
            webpage_url = ytdlp_info["url"]
            fallback_meta = ytdlp_info
            LOGGER.info(f"✅ yt-dlp found: {fallback_meta['title']} ({fallback_meta['duration']}s)")
        else:
            LOGGER.warning("⚠️ yt-dlp search failed, using OneGrab fallback.")
            search_data = await _api_request("/api/search", {"query": query, "platform": "youtube"})
            if search_data and search_data.get("results"):
                track = search_data["results"][0]
                webpage_url = track.get("url")
                fallback_meta = {"title": track.get("title", query), "duration": int(track.get("duration") or 0), "thumbnail": track.get("thumbnail", "")}

    if not webpage_url: return None

    # ✅ 2. OneGrab Stream Link
    req_type = "video" if video else "audio"
    LOGGER.info(f"🌐 Fetching {req_type} stream via OneGrab...")
    track_data = await _api_request("/api/track", {"url": webpage_url, "type": req_type})
    
    if track_data:
        result = track_data
        if "results" in track_data and isinstance(track_data["results"], list) and len(track_data["results"]) > 0:
            result = track_data["results"][0]

        cdnurl = result.get("cdnurl") or result.get("download_url") or result.get("audio_url") or result.get("video_url")
        
        if cdnurl:
            # ✅ ALWAYS prioritize yt-dlp metadata
            final_thumb = fallback_meta.get("thumbnail") or result.get("thumbnail", "")
            final_dur = fallback_meta.get("duration") or int(result.get("duration") or 0)
            final_title = fallback_meta.get("title") or result.get("title", "Unknown")
            
            if "t.me" in cdnurl or "telegram.me" in cdnurl:
                local_path = await _download_telegram_cdn(cdnurl)
                if local_path:
                    return {"title": final_title, "url": local_path, "duration": final_dur, "thumbnail": final_thumb, "webpage_url": webpage_url}
            else:
                return {"title": final_title, "url": cdnurl, "duration": final_dur, "thumbnail": final_thumb, "webpage_url": webpage_url}
    return None

import aiohttp
import re
import asyncio
from urllib.parse import quote
from pyrogram.errors import FloodWait
from SunloMusic.logger import LOGGER

async def _api_get(url, headers, timeout=10):
    try:
        conn = aiohttp.TCPConnector(family=2) # Force IPv4
        async with aiohttp.ClientSession(headers=headers, connector=conn) as s:
            async with s.get(url, timeout=timeout) as r:
                if r.status == 200: return await r.json()
    except Exception as e: LOGGER.error(f"API Error: {e}")
    return None

def _parse_song_info(query: str):
    clean = re.sub(r'[\(\[].*?[\)\]]', '', query)
    parts = [p.strip() for p in re.split(r'[|\-]', clean) if p.strip()]
    if not parts: return query, ""
    title = parts[0]
    artist = parts[1] if len(parts) > 1 else ""
    if " by " in title.lower():
        t_parts = re.split(r' by ', title, flags=re.IGNORECASE)
        title = t_parts[0].strip()
        artist = t_parts[1].strip() if len(t_parts) > 1 else artist
    return title.strip(), artist.strip()

async def fetch_lyrics_text(query: str, duration: int = 0) -> str | None:
    title, artist = _parse_song_info(query)
    headers = {"User-Agent": "SunloMusic/1.0"}
    
    # LRCLIB
    try:
        url = f"https://lrclib.net/api/get?track_name={quote(title)}"
        if artist: url += f"&artist_name={quote(artist)}"
        if duration > 0: url += f"&duration={duration}"
        data = await _api_get(url, headers)
        if data and "plainLyrics" in data and data["plainLyrics"]: return data["plainLyrics"]
        
        search_url = f"https://lrclib.net/api/search?q={quote(title)}"
        if artist: search_url += f"+{quote(artist)}"
        search_data = await _api_get(search_url, headers)
        if search_data and isinstance(search_data, list) and len(search_data) > 0:
            for track in search_data:
                if "plainLyrics" in track and track["plainLyrics"]: return track["plainLyrics"]
    except: pass

    # JioSaavn
    try:
        search_url = f"https://saavnapi-nine.vercel.app/result/?query={quote(title)}"
        data = await _api_get(search_url, headers)
        if data and "songs" in data and "results" in data["songs"] and len(data["songs"]["results"]) > 0:
            lyrics_url = f"https://saavnapi-nine.vercel.app/lyrics/?id={data['songs']['results'][0]['id']}"
            lyrics_data = await _api_get(lyrics_url, headers)
            if lyrics_data and "lyrics" in lyrics_data and lyrics_data["lyrics"]:
                lyrics = re.sub(r'<br\s*/?>', '\n', lyrics_data["lyrics"])
                return re.sub(r'<.*?>', '', lyrics).strip()
    except: pass

    # Lyrics.ovh
    if artist and title:
        try:
            url = f"https://api.lyrics.ovh/v1/{quote(artist)}/{quote(title)}"
            data = await _api_get(url, headers)
            if data and "lyrics" in data and data["lyrics"]: return data["lyrics"]
        except: pass

    return None

async def start_live_karaoke(bot, chat_id, title, duration, q):
    msg = await bot.send_message(chat_id, f"🎤 <b>Fetching live lyrics for:</b> <i>{title}</i>...")
    
    try:
        lyrics = await asyncio.wait_for(fetch_lyrics_text(title, duration), timeout=20.0)
        
        if not lyrics:
            await msg.edit_text(f"🎤 <b>Karaoke:</b> Lyrics not found for <i>{title}</i>.")
            return

        lines = [line.strip() for line in lyrics.split('\n') if line.strip()]
        if not lines: return

        # ✅ Fixed 5 second delay as requested
        delay = 5.0
        header = "🎤 <b>Live Karaoke</b>\n\n<blockquote>"
        accumulated = lines[0]
        
        await msg.edit_text(f"{header}{accumulated}</blockquote>")
        q.karaoke_msg_id = msg.id

        for i in range(1, len(lines)):
            if not q.now_playing or q.now_playing.title != title or q.paused: break
            
            await asyncio.sleep(delay)
            next_line = lines[i]
            
            if len(accumulated) + len(next_line) + len(header) > 4000:
                try: await msg.edit_text(f"{header}{accumulated}</blockquote>")
                except: pass
                accumulated = next_line
                msg = await bot.send_message(chat_id, f"{header}{accumulated}</blockquote>")
                q.karaoke_msg_id = msg.id
            else:
                accumulated += f"\n{next_line}"
                try:
                    await msg.edit_text(f"{header}{accumulated}</blockquote>")
                except FloodWait as e:
                    await asyncio.sleep(e.value)
                    try: await msg.edit_text(f"{header}{accumulated}</blockquote>")
                    except: pass
                except Exception:
                    pass

    except asyncio.TimeoutError:
        await msg.edit_text("❌ <b>Karaoke Error:</b> Timed out.")
    except asyncio.CancelledError: pass
    except Exception as e:
        try: await msg.edit_text(f"❌ <b>Karaoke Error:</b> <code>{e}</code>")
        except: pass
